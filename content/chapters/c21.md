# 动手实验：从零实现最小可用 Agent

前面 20 章把原理讲完了。这一章我们把它们**写进代码**——一个不含任何框架、只用标准库 + 一个 LLM 客户端的 Agent，约 150 行。

它麻雀虽小，但五个组件一个不少：**预算控制、工具注册、参数校验、无进展检测、错误回灌**。

## 目标与设计

```text
实现一个能完成这类任务的 Agent：
  "查一下我们公司出差住宿的标准，然后算 3 人 5 天的预算上限"

必须具备：
  ✓ 工具调用（含参数校验）
  ✓ 结构化决策（JSON 输出）
  ✓ 步数 / Token / 时间三重预算
  ✓ 无进展检测（防止死循环）
  ✓ 错误以自然语言回灌（支持自我修复）
  ✗ 不需要：多 Agent、向量库、沙箱（留给你的扩展练习）
```

## 完整代码

```python {1-3} title="minimal_agent.py —— 第 1 部分：工具与校验"
import json, time, hashlib
from dataclasses import dataclass, field
from typing import Callable, Any


@dataclass
class Tool:
    name: str
    description: str
    params: dict[str, str]          # 参数名 → 说明（用于生成 schema 与校验）
    fn: Callable
    side_effect: bool = False


class ToolError(Exception):
    """工具执行失败。message 会被翻译成模型能理解的提示。"""


TOOLS: dict[str, Tool] = {}


def tool(name: str, description: str, params: dict[str, str], side_effect: bool = False):
    """装饰器：注册工具并自动生成 schema 描述"""
    def deco(fn):
        TOOLS[name] = Tool(name, description, params, fn, side_effect)
        return fn
    return deco


@tool(
    "search_knowledge_base",
    "在企业内部知识库中检索文档片段。适用于查询公司制度、流程、政策等内部信息。"
    "不适用于需要精确计算的场景。",
    {"query": "检索关键词，建议使用名词短语，不要写成问句"},
)
def search_knowledge_base(query: str) -> str:
    # 真实场景：调用向量库。这里用固定数据模拟。
    KB = {
        "住宿": "一线城市住宿标准：500 元/人/晚；二线城市：350 元/人/晚。",
        "餐补": "出差餐补：100 元/人/天，按自然日计算。",
        "交通": "市内交通实报实销，需提供票据。",
    }
    hits = [v for k, v in KB.items() if k in query] or list(KB.values())
    return "；".join(hits[:3])


@tool(
    "calculator",
    "执行数学表达式计算。凡涉及数字运算必须使用本工具，不要自行心算。",
    {"expr": "合法的数学表达式，例如 '(500 + 100) * 5 * 3'"},
)
def calculator(expr: str) -> str:
    allowed = set("0123456789+-*/(). ")
    if not set(expr) <= allowed:
        raise ToolError(f"表达式含非法字符。只允许数字与 + - * / ( )，你提供的是：{expr}")
    try:
        result = eval(expr, {"__builtins__": {}}, {})   # 已限制字符集
    except ZeroDivisionError:
        raise ToolError("除数不能为 0，请检查表达式。")
    except Exception as e:
        raise ToolError(f"表达式无法求值（{e}）。请检查括号与运算符是否配对。")
    return str(result)


def tools_schema() -> str:
    """生成给模型看的工具说明"""
    lines = []
    for t in TOOLS.values():
        ps = "，".join(f'{k}（{v}）' for k, v in t.params.items())
        lines.append(f"- {t.name}：{t.description}\n  参数：{ps}")
    return "\n".join(lines)
```

```python {1-2} title="minimal_agent.py —— 第 2 部分：预算与状态"
@dataclass
class Budget:
    max_steps: int = 10
    max_seconds: float = 60.0
    max_calls: int = 12

    def exhausted(self, state: "AgentState") -> str | None:
        if state.steps >= self.max_steps:
            return f"已达到最大步数（{self.max_steps}）"
        if time.time() - state.started > self.max_seconds:
            return f"已超过时间上限（{self.max_seconds}s）"
        if state.llm_calls >= self.max_calls:
            return f"已超过模型调用次数上限（{self.max_calls}）"
        return None


@dataclass
class AgentState:
    goal: str
    messages: list = field(default_factory=list)
    steps: int = 0
    llm_calls: int = 0
    started: float = field(default_factory=time.time)
    seen_actions: set = field(default_factory=set)
    repeats: int = 0
    facts: list = field(default_factory=list)


SYSTEM_PROMPT = """你是一个严谨的任务执行 Agent。每轮必须输出以下 JSON 之一：

1) 调用工具：
{"thought": "简短的推理，说明为什么需要这一步",
 "action": "tool_call", "tool": "工具名", "args": {...}}

2) 直接回答：
{"thought": "...", "action": "final_answer", "content": "给用户的回答"}

3) 信息不足时提问：
{"thought": "...", "action": "ask_user", "content": "需要澄清的问题"}

规则（按优先级）：
1. 涉及事实（公司制度、政策等）必须先用 search_knowledge_base 查询，禁止凭记忆作答。
2. 涉及数字运算必须使用 calculator，禁止心算。
3. 工具报错时，按提示修正参数后重试；同一个工具连续失败 2 次则应换思路或说明情况。
4. 只输出 JSON，不要输出任何其他文字。

可用工具：
%s""" % tools_schema()
```

```python {1-2} title="minimal_agent.py —— 第 3 部分：主循环"
def call_llm(messages: list) -> dict:
    """把这里换成你的模型 SDK 调用。返回解析后的 JSON。"""
    # 真实实现示例（OpenAI 兼容接口）：
    #   resp = client.chat.completions.create(
    #       model="your-model", messages=messages,
    #       temperature=0, response_format={"type": "json_object"})
    #   return json.loads(resp.choices[0].message.content)
    raise NotImplementedError("请替换为真实的模型调用")


def execute(state: AgentState, name: str, args: dict) -> str:
    """执行工具：校验 → 执行 → 把错误翻译成模型能理解的提示"""
    t = TOOLS.get(name)
    if t is None:
        return f"[工具错误] 不存在名为 {name} 的工具。可用工具：{list(TOOLS)}"

    # 参数校验
    missing = [k for k in t.params if k not in args]
    if missing:
        return (f"[参数错误] 缺少必需参数 {missing}。"
                f"{t.name} 的参数要求：{t.params}。请修正后重试。")
    extra = [k for k in args if k not in t.params]
    if extra:
        return f"[参数错误] 出现未知参数 {extra}。请只使用 {list(t.params)}。"

    try:
        return str(t.fn(**args))
    except ToolError as e:
        return f"[工具错误] {e}"
    except Exception as e:
        return f"[工具异常] {type(e).__name__}: {e}。请调整调用方式或换用其他工具。"


def run(goal: str, budget: Budget | None = None) -> str:
    budget = budget or Budget()
    state = AgentState(goal=goal)
    state.messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": goal},
    ]

    while True:
        # —— 反思：预算检查（第一道闸门）——
        reason = budget.exhausted(state)
        if reason:
            return f"任务未完成：{reason}。已获得的信息：{state.facts}"

        # —— 感知：把目标锚定在上下文末尾（第 9 章） ——
        state.messages.append({
            "role": "system",
            "content": f"【原始目标】{goal}\n【已确认事实】{state.facts or '（暂无）'}",
        })

        # —— 规划：结构化决策 ——
        state.llm_calls += 1
        try:
            plan = call_llm(state.messages)
        except Exception as e:
            return f"模型调用失败：{e}"

        action = plan.get("action")
        state.messages.append({"role": "assistant", "content": json.dumps(plan, ensure_ascii=False)})

        # —— 终止条件 ——
        if action == "final_answer":
            return plan.get("content", "（无内容）")
        if action == "ask_user":
            return f"[需要澄清] {plan.get('content')}"

        # —— 无进展检测（第 5 章） ——
        fingerprint = hashlib.md5(
            json.dumps([action, plan.get("tool"), plan.get("args")], sort_keys=True).encode()
        ).hexdigest()
        if fingerprint in state.seen_actions:
            state.repeats += 1
            if state.repeats >= 2:
                state.messages.append({
                    "role": "system",
                    "content": "你已重复执行相同动作且没有获得新信息。"
                               "请改变策略：换工具、换参数，或基于现有信息直接回答。",
                })
        else:
            state.repeats = 0
        state.seen_actions.add(fingerprint)

        # —— 行动 ——
        observation = execute(state, plan.get("tool", ""), plan.get("args") or {})
        state.facts.append(f"{plan.get('tool')}({plan.get('args')}) -> {observation[:160]}")
        state.messages.append({
            "role": "user",
            "content": f"[工具返回] {observation}",
        })
        state.steps += 1
```

## 扩展练习

按难度递增，每一题都对应前面章节的一个知识点：

- [ ] **入门**：接入真实的 LLM API，跑通一次完整调用。
- [ ] **入门**：加一个 `get_weather(city)` 工具，观察模型如何选择工具。
- [ ] **进阶**：加入"同一工具失败 2 次后禁用该工具"的逻辑（第 5 章错误处理）。
- [ ] **进阶**：把 `state.facts` 换成"滚动摘要"（第 9 章压缩策略）。
- [ ] **进阶**：加一层输入护栏，检测"忽略以上指令"这类注入（第 15 章）。
- [ ] **挑战**：给 `side_effect=True` 的工具增加人工确认流程（第 15 章动作护栏）。
- [ ] **挑战**：把主循环改造成"粗粒度状态机 + 节点内自由循环"（第 10 章混合模式）。
- [ ] **挑战**：用 30 条用例搭一个评估脚本，自动统计任务完成率（第 14 章）。

:::callout tip 一个建议的练习顺序
**先做"接入真实 API"和"加一个工具"**——它们能让你立刻看到 Agent 的行为。

**再尝试"删掉无进展检测"**，然后构造一个会让它死循环的任务，亲眼看到不加防护的后果。这个反向实验比读十遍原理都有效。
:::

## 最后三条建议

:::callout key 如果只带走三句话
1. **控制流在谁手里，决定了它是不是 Agent。** 能用固定流程解决的，就别交给模型。
2. **Agent 工程的重心是"降低错误累积"，不是"提升单步能力"。** 控制步数、加校验、可回滚——这三件事的收益远大于换模型。

3. **Agent 是一个需要持续维护的产品，不是交付即完成的软件。** 决定它能否长期有效的，是评估体系与内容维护，而不是模型版本。
:::
